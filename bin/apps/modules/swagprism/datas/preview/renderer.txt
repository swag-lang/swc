using Core, Pixel

#main
{
    Swag.setWorkerCount(6)
    var channel: Pipeline.FrameChannel
    try channel.open(Env.arguments()[1])
    let header = channel.header()
    var clock: Time.Stopwatch
    clock.start()
    var frame = 0'u32
    for
    {
        channel.waitConsumed()
        var watch: Time.Stopwatch
        watch.start()
        let pixels = cast([*] u32) channel.pixels()
        let time = frame == 0 ? 0.0 : cast(f32) clock.elapsedMicroseconds() / 1000000
        parallel for |pixels, time| y in cast(u64) Pipeline.FrameChannel.Height
        {
            for x in Pipeline.FrameChannel.Width do
                pixels[y * Pipeline.FrameChannel.Width + x] = shade(
                    (cast(f32) x - 320) / 200, (cast(f32) y - 200) / 200, time)
        }
        header.renderUs = cast(u32) Math.min(watch.elapsedMicroseconds(), Swag.U32.Max)
        frame += 1
        Atomic.exchange(&header.published, frame)
    }
}
